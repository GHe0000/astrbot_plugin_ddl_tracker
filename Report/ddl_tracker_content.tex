\section{问题与目标}

\begin{frame}
  \frametitle{待解决问题：Q 群通知过多}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{现实痛点}
        \begin{itemize}
          \item Q 群里课程、考试、报名、作业通知混在一起，DDL 容易被刷走。
          \item DDL 表达不统一：绝对时间、相对时间、口语化提醒规则同时存在。
          \item 手动从聊天切到日历创建提醒，步骤多、打断强、容易漏。
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{noteBlock}
        目标是把“看到一条群消息”变成“可追踪、可提醒、可恢复的日程任务”。
      \end{noteBlock}
      \vspace{0.25cm}
      \begin{itemize}
        \item 通过用户手动转发消息给 Agent，降低创建日程的交互成本。
        \item 通过 AstrBot FutureTask 创建一次性 cron 提醒。
        \item 通过插件状态记录避免重复创建和重启丢失。
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

    \begin{column}{0.32\textwidth}
      \alert{可识别}\\
      从自然语言里提取标题、类型、截止时间、原始文本。
    \end{column}
    \begin{column}{0.32\textwidth}
      \alert{可去重}\\
      用标题和时间生成指纹，避免相同 DDL 反复创建任务。
    \end{column}
    \begin{column}{0.32\textwidth}
      \alert{可回写}\\
      FutureTask 创建后把任务名写回 DDL 状态，支持查询和恢复。
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{主要运行思路：后台扫描与任务同步}
  \begin{center}
  \begin{tikzpicture}[node distance=0.42cm, >=Latex, font=\scriptsize]
    \tikzstyle{box}=[draw=black!25, fill=black!4, rounded corners=2pt, minimum width=2.05cm, minimum height=0.78cm, align=center]
    \node[box] (m) {消息缓存};
    \node[box, right=of m] (loop) {定时扫描};
    \node[box, right=of loop] (llm) {LLM 提取};
    \node[box, right=of llm] (merge) {合并去重};
    \node[box, right=of merge] (plan) {提醒计划};
    \node[box, below=0.95cm of plan] (pending) {pending tasks};
    \node[box, left=of pending] (future) {FutureTask};
    \node[box, left=of future] (mark) {状态回写};

    \draw[-Latex, thick] (m) -- (loop);
    \draw[-Latex, thick] (loop) -- (llm);
    \draw[-Latex, thick] (llm) -- (merge);
    \draw[-Latex, thick] (merge) -- (plan);
    \draw[-Latex, thick] (plan) -- (pending);
    \draw[-Latex, thick] (pending) -- (future);
    \draw[-Latex, thick] (future) -- (mark);
  \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item 当前主 Agent 工具链已能自动同步 FutureTask。
    \item 后台扫描负责生成 pending 状态，提醒任务由 FutureTask 统一调度。
    \item 失败时保留 pending 状态，下次扫描继续重试，避免任务丢失。
  \end{itemize}
\end{frame}

\section{接口与成果}

\begin{frame}
  \frametitle{用户接口与 AI 接口}
  \small
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{block}{用户接口}
        \begin{itemize}\setlength{\itemsep}{1pt}
          \item \texttt{/ddl\_on}: 开启群 DDL 跟踪。
          \item \texttt{/ddl\_extract}: 手动整理最近消息。
          \item \texttt{/ddl\_status}: 查看 DDL、规则、任务状态。
          \item \texttt{/ddl\_nearest}: 查看最近截止事项。
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.51\textwidth}
      \begin{block}{AI Function Calling}
        \begin{itemize}\setlength{\itemsep}{1pt}
          \item \texttt{ddl\_extract\_recent\_messages}
          \item \texttt{ddl\_set\_type\_reminder}
          \item \texttt{ddl\_get\_pending\_future\_tasks}
          \item \texttt{ddl\_mark\_future\_task\_created}
          \item \texttt{future\_task}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.05cm}
  \begin{noteBlock}
    \scriptsize pending 任务携带 \texttt{future\_task\_arguments}；创建成功后用 \texttt{mark\_created\_arguments} 回写绑定状态。
  \end{noteBlock}
\end{frame}
